// AssignmentTask06 : Substraction of Nodes

    public static Integer subtractSummation( BTNode root ){

        if(root == null) {
            return 0;
        }

        return signedWalk(root.left, 1) + signedWalk(root.right, -1);
    }

    private static int signedWalk(BTNode node, int polarity) {

        if(node == null) {
            return 0;
        }

        int ojon = polarity * (Integer) node.elem;

        return ojon + signedWalk(node.left, polarity) + signedWalk(node.right, polarity);
    }

// AssignmentTask07 : Difference of Level Sum

    public static Integer levelSum( BTNode root ){

        return alternatingWalk(root, -1);
    }

    private static int alternatingWalk(BTNode node, int polarity) {

        if(node == null) {
            return 0;
        }

        int ojon = polarity * (Integer) node.elem;
        int niche = alternatingWalk(node.left, -polarity) + alternatingWalk(node.right, -polarity);

        return ojon + niche;
    }

// AssignmentTask08 : Swap Children Nodes

    public static void swapChild( BTNode root, int lvl, int M ){

        mirrorTop(root, M - lvl);
    }

    private static void mirrorTop(BTNode node, int budget) {

        if(node == null || budget <= 0) {
            return;
        }

        BTNode dhore_rakhe = node.left;
        node.left = node.right;
        node.right = dhore_rakhe;

        mirrorTop(node.left, budget - 1);
        mirrorTop(node.right, budget - 1);
    }

// AssignmentTask04 : Sum of Range

    public static Integer rangeSum( BSTNode root, Integer low, Integer high ){

        if(root == null || low == null || high == null) {
            return 0;
        }

        return window(root, low, high);
    }

    private static int window(BSTNode node, int floor, int ceil) {

        if(node == null || floor > ceil) {
            return 0;
        }
            
        int shongkha = (Integer) node.elem;

        if(shongkha < floor) {
            return window(node.right, floor, ceil);
        }

        if(shongkha > ceil) {
            return window(node.left, floor, ceil);
        }

        return shongkha + window(node.left, floor, shongkha) + window(node.right, shongkha, ceil);
    }
